/**
 * VehicleService - Service Layer for Vehicle Management
 * Implements business logic for vehicle operations
 * Uses Dependency Injection for better testability and modularity
 */
class VehicleService {
  constructor(vehicleModel, bookingModel) {
    this.vehicleModel = vehicleModel;
    this.bookingModel = bookingModel;
  }

  async addVehicle(vehicleData) {
    try {
      const {
        Vehicletype,
        model,
        pricePerDay,
        capacity,
        licensePlate,
        images,
      } = vehicleData;

      // Check for required fields
      if (!Vehicletype || !model || !pricePerDay || !capacity) {
        throw new Error(
          "Missing required fields: Vehicletype, model, pricePerDay, or capacity."
        );
      }

      // Check if a vehicle with the same license plate already exists
      if (licensePlate) {
        const existingVehicle = await this.vehicleModel.findOne({ licensePlate });
        if (existingVehicle) {
          throw new Error("A vehicle with this license plate already exists.");
        }
      }

      // Create a new vehicle instance
      const newVehicle = new this.vehicleModel({
        Vehicletype,
        model,
        pricePerDay,
        capacity,
        licensePlate,
        images: images ?? null,
      });

      // Save the vehicle to the database
      const savedVehicle = await newVehicle.save();

      return savedVehicle;
    } catch (error) {
      console.error("Error adding vehicle:", error.message);
      throw new Error(`Failed to add vehicle: ${error.message}`);
    }
  }

  async getAllVehicles() {
    try {
      return await this.vehicleModel.find();
    } catch (error) {
      throw new Error(`Failed to fetch vehicles: ${error.message}`);
    }
  }

  async getVehicleById(id) {
    try {
      const vehicle = await this.vehicleModel.findById(id);

      if (!vehicle) {
        throw new Error("Vehicle not found");
      }

      return vehicle;
    } catch (error) {
      throw new Error(`Failed to fetch vehicle by ID: ${error.message}`);
    }
  }

  async updateVehicle(id, updateData) {
    try {
      const updatedVehicle = await this.vehicleModel.findByIdAndUpdate(
        id,
        updateData,
        { new: true, runValidators: true }
      );

      if (!updatedVehicle) {
        throw new Error("Vehicle not found");
      }

      return updatedVehicle;
    } catch (error) {
      throw new Error(`Failed to update vehicle: ${error.message}`);
    }
  }

  async removeVehicle(id) {
    try {
      const deletedVehicle = await this.vehicleModel.findByIdAndDelete(id);

      if (!deletedVehicle) {
        throw new Error("Vehicle not found");
      }

      return deletedVehicle;
    } catch (error) {
      throw new Error(`Failed to remove vehicle: ${error.message}`);
    }
  }

  async getAvailableVehicles(startDate, endDate) {
    const start = new Date(startDate);
    const end = new Date(endDate);

    const conflictingBookings = await this.bookingModel.find({
      $or: [{ startDate: { $lt: end }, endDate: { $gt: start } }],
    }).select("vehicle");

    const bookedVehicleIds = conflictingBookings.map(
      (booking) => booking.vehicle
    );

    const availableVehicles = await this.vehicleModel.find({
      _id: { $nin: bookedVehicleIds },
      availability: true,
    });

    return availableVehicles;
  }

  async getVehiclesByLocation(latitude, longitude, radiusInKm) {
    try {
      // Calculate bounding box
      const earthRadiusKm = 6371;
      const latDelta = (radiusInKm / earthRadiusKm) * (180 / Math.PI);
      const lonDelta = (radiusInKm / (earthRadiusKm * Math.cos((latitude * Math.PI) / 180))) * (180 / Math.PI);

      const vehicles = await this.vehicleModel.find({
        'location.latitude': {
          $gte: latitude - latDelta,
          $lte: latitude + latDelta,
        },
        'location.longitude': {
          $gte: longitude - lonDelta,
          $lte: longitude + lonDelta,
        },
        availability: true,
      });

      // Calculate actual distance and add to results
      const vehiclesWithDistance = vehicles.map(vehicle => {
        const distance = this.calculateDistance(
          latitude,
          longitude,
          vehicle.location.latitude,
          vehicle.location.longitude
        );
        return {
          ...vehicle.toObject(),
          distanceKm: distance,
        };
      });

      // Sort by distance
      vehiclesWithDistance.sort((a, b) => a.distanceKm - b.distanceKm);

      return vehiclesWithDistance;
    } catch (error) {
      throw new Error(`Error getting vehicles by location: ${error.message}`);
    }
  }

  calculateDistance(lat1, lon1, lat2, lon2) {
    const R = 6371; // Earth radius in km
    const dLat = this.toRad(lat2 - lat1);
    const dLon = this.toRad(lon2 - lon1);
    const a =
      Math.sin(dLat / 2) * Math.sin(dLat / 2) +
      Math.cos(this.toRad(lat1)) * Math.cos(this.toRad(lat2)) *
      Math.sin(dLon / 2) * Math.sin(dLon / 2);
    const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    return R * c;
  }

  toRad(deg) {
    return deg * (Math.PI / 180);
  }
}

module.exports = VehicleService;
